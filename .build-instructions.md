The source videos are 1280×720 from Sora. Upscale to 1920×1080 with sharpening:

apt-get install -y ffmpeg 2>/dev/null || true
for f in public/videos/*.mp4; do
  ffmpeg -y -i "$f" \
    -vf "scale=1920:1080:flags=lanczos,unsharp=5:5:1.2:5:5:0.5,eq=contrast=1.08:saturation=1.12:brightness=0.02" \
    -c:v libx264 -crf 18 -preset medium -movflags +faststart \
    -c:a copy "${f%.mp4}_hq.mp4" && mv "${f%.mp4}_hq.mp4" "$f"
done
echo "Video enhancement done"
